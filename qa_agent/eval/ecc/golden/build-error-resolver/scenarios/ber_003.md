# Scenario: Generic Type Does Not Satisfy Constraint

## Code File: `src/utils/repository.ts`

```typescript
interface Identifiable {
  id: string;
  createdAt: Date;
}

class Repository<T extends Identifiable> {
  private items: T[] = [];

  add(item: T): void {
    this.items.push(item);
  }

  findById(id: string): T | undefined {
    return this.items.find((item) => item.id === id);
  }

  findRecent(since: Date): T[] {
    return this.items.filter((item) => item.createdAt > since);
  }
}

// Usage that triggers the error
interface SimpleTag {
  id: string;
  label: string;
}

const tagRepo = new Repository<SimpleTag>();
tagRepo.add({ id: "t1", label: "camping" });
```

## Build Command

```bash
npx tsc --noEmit
```

## Error Output

```
src/utils/repository.ts:26:27 - error TS2344: Type 'SimpleTag' does not satisfy the
constraint 'Identifiable'.
  Property 'createdAt' is missing in type 'SimpleTag' but required in type 'Identifiable'.

26 const tagRepo = new Repository<SimpleTag>();
                             ~~~~~~~~~
```

## Task

Fix the build error. The `SimpleTag` type does not have a `createdAt` field, but `Repository` requires it via the `Identifiable` constraint. Choose the best fix: either add `createdAt` to `SimpleTag`, relax the constraint, or split the repository into two variants.
