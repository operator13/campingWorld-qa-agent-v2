# Scenario: Missing Required Property in Object Literal

## Code File: `src/models/product.ts`

```typescript
interface Product {
  id: string;
  name: string;
  price: number;
  category: string;
  inStock: boolean;
}

function createProduct(name: string, price: number, category: string): Product {
  return {
    id: crypto.randomUUID(),
    name,
    price,
    category,
  };
}

export { createProduct, Product };
```
