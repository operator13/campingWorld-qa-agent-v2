# Scenario: Circular Dependency Between Two Modules

## Code File 1: `src/models/user.ts`

```typescript
import { Order } from "./order";

export interface User {
  id: string;
  name: string;
  email: string;
  orders: Order[];
}

export function getUserDisplayName(user: User): string {
  return `${user.name} (${user.email})`;
}

export function getUserOrderTotal(user: User): number {
  return user.orders.reduce((sum, order) => sum + order.total, 0);
}
```

## Code File 2: `src/models/order.ts`

```typescript
import { User, getUserDisplayName } from "./user";

export interface Order {
  id: string;
  total: number;
  status: string;
  placedBy: User;
}

export function getOrderLabel(order: Order): string {
  return `Order ${order.id} by ${getUserDisplayName(order.placedBy)} - $${order.total}`;
}
```

## Build Command

```bash
npx tsc --noEmit
```

## Error Output

```
error TS2306: File 'src/models/user.ts' is not a module.

The file is part of a circular dependency:
  src/models/user.ts -> src/models/order.ts -> src/models/user.ts

This can cause runtime errors where imports resolve to undefined.
```

## Task

Break the circular dependency. `user.ts` imports `Order` from `order.ts`, and `order.ts` imports `User` and `getUserDisplayName` from `user.ts`. Extract the shared interfaces into a `types.ts` file or restructure the imports to eliminate the cycle.
