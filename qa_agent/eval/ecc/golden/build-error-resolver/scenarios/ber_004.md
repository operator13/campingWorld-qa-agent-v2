# Scenario: Wrong Relative Import Path

## Project Structure

```
src/
  components/
    Header.tsx
  utils/
    formatters.ts
  pages/
    ProductPage.tsx
```

## Code File: `src/pages/ProductPage.tsx`

```typescript
import { Header } from "../components/Header";
import { formatPrice } from "./utils/formatters";

interface ProductPageProps {
  name: string;
  price: number;
}

export function ProductPage({ name, price }: ProductPageProps) {
  return {
    header: Header({ title: name }),
    priceDisplay: formatPrice(price),
  };
}
```

## Build Command

```bash
npx tsc --noEmit
```

## Error Output

```
src/pages/ProductPage.tsx:2:30 - error TS2307: Cannot find module './utils/formatters'
or its corresponding type declarations.

2 import { formatPrice } from "./utils/formatters";
                               ~~~~~~~~~~~~~~~~~~~~
```

## Task

Fix the import path. The file `formatters.ts` is in `src/utils/`, not `src/pages/utils/`. The correct relative path from `src/pages/` to `src/utils/` is `../utils/formatters`.
