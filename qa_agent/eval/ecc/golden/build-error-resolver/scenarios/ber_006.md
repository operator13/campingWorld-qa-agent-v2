# Scenario: Package Not Installed

## Code File: `src/api/client.ts`

```typescript
import axios from "axios";
import { z } from "zod";

const ProductSchema = z.object({
  id: z.string(),
  name: z.string(),
  price: z.number().positive(),
});

type Product = z.infer<typeof ProductSchema>;

async function fetchProducts(baseUrl: string): Promise<Product[]> {
  const response = await axios.get(`${baseUrl}/api/products`);
  const parsed = z.array(ProductSchema).parse(response.data);
  return parsed;
}

export { fetchProducts, Product };
```

## package.json (relevant section)

```json
{
  "dependencies": {
    "axios": "^1.6.0"
  },
  "devDependencies": {
    "@types/node": "^20.0.0",
    "typescript": "^5.3.0"
  }
}
```

## Task

Identify that `zod` is used in the source code but is not listed in `package.json` dependencies. The fix is to run `npm install zod` (runtime dependency, not dev dependency, since it is used in production code for runtime validation).
