```typescript
// page_objects/WishlistPage.ts
import { Page, Locator } from '@playwright/test';

export class WishlistPage {
  readonly page: Page;
  readonly wishlistItems: Locator;
  readonly emptyMessage: Locator;
  readonly removeButton: (name: string) => Locator;

  constructor(page: Page) {
    this.page = page;
    this.wishlistItems = page.getByTestId('wishlist-item');
    this.emptyMessage = page.getByText('Your wishlist is empty');
    this.removeButton = (name: string) =>
      page.getByRole('button', { name: `Remove ${name}` });
  }

  async goto() { await this.page.goto('/account/wishlist'); }
  async removeItem(name: string) { await this.removeButton(name).click(); }
  async itemCount() { return this.wishlistItems.count(); }
}
```

```typescript
// tests_generated/wishlist.spec.ts
import { test, expect } from '@playwright/test';
import { WishlistPage } from '../page_objects/WishlistPage';

test.describe('Wishlist', () => {
  test('shows empty state when no items saved', async ({ page }) => {
    const wishlist = new WishlistPage(page);
    await wishlist.goto();

    await expect(wishlist.emptyMessage).toBeVisible();
    expect(await wishlist.itemCount()).toBe(0);
  });
});
```
