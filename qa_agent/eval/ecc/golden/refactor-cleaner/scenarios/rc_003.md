# Scenario: Exported type only used internally

## Task
Analyze `services/cache.ts` for unnecessary exports.

## File: services/cache.ts
```typescript
export interface CacheConfig {
  ttlMs: number;
  maxSize: number;
}

export interface InternalCacheEntry {
  key: string;
  value: unknown;
  expiresAt: number;
}

export class CacheService {
  private entries: Map<string, InternalCacheEntry> = new Map();
  private config: CacheConfig;

  constructor(config: CacheConfig) {
    this.config = config;
  }

  get(key: string): unknown | undefined {
    const entry = this.entries.get(key);
    if (!entry || entry.expiresAt < Date.now()) return undefined;
    return entry.value;
  }

  set(key: string, value: unknown): void {
    const entry: InternalCacheEntry = {
      key, value, expiresAt: Date.now() + this.config.ttlMs
    };
    this.entries.set(key, entry);
  }
}
```

## Codebase usage (grep results)
```
src/app.ts:           import { CacheService, CacheConfig } from './services/cache';
tests/cache.test.ts:  import { CacheService, CacheConfig } from './services/cache';
```

## Expected Finding
`InternalCacheEntry` is exported but only used within `cache.ts` itself.
The `export` keyword should be removed. `CacheConfig` is used externally
and must remain exported.
