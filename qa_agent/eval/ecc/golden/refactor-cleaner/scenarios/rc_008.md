import re

def sanitize_html(raw: str) -> str:
    """Strip HTML tags and normalize whitespace."""
    text = re.sub(r"<[^>]+>", "", raw)
    text = re.sub(r"\s+", " ", text)
    return text.strip()

def escape_sql_like(value: str) -> str:
    """Escape special chars for SQL LIKE patterns."""
    return value.replace("%", "\\%").replace("_", "\\_")
